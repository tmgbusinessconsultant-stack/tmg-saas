  private projectInclude() {
    return {
      User_Project_ownerIdToUser: { select: { id: true, name: true, email: true } },
      User_Project_assigneeIdToUser: { select: { id: true, name: true, email: true } },
      ProjectFile: { 
        orderBy: { createdAt: 'desc' as const }, 
        select: { id: true, name: true, originalName: true, contentType: true, size: true, storageKey: true, kind: true, status: true, createdAt: true } 
      },
      ProjectMessage: { 
        orderBy: { createdAt: 'desc' as const }, 
        select: { id: true, body: true, createdAt: true, User: { select: { id: true, name: true } } } 
      },
      ProjectActivity: { 
        orderBy: { createdAt: 'desc' as const }, 
        select: { id: true, action: true, detail: true, createdAt: true } 
      },
    };
  }

  private publicProject(project: any) {
    const files = project.ProjectFile || project.files || [];
    return {
      ...project,
      files: files.map((f: any) => ({ ...f, size: f.size?.toString() })),
      owner: project.User_Project_ownerIdToUser || project.owner,
      assignee: project.User_Project_assigneeIdToUser || project.assignee,
      messages: project.ProjectMessage || project.messages || [],
      activities: project.ProjectActivity || project.activities || [],
    };
  }
